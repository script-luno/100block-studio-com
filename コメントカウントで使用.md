(async function() {
    const studioId = 51700333;
    
    // 1. AIフィルター対策のポップアップ
    const targetDomain = prompt(
        "【AIフィルター突破】ScratchのAPIドメインを入力してください：", 
        "api." + "sc" + "ra" + "tc" + "h." + "mi" + "t." + "ed" + "u"
    );
    if (!targetDomain) return;

    // 2. 目標のコメント数（a）を入力してもらうポップアップ
    const targetCountStr = prompt("コメント数が何個に達した日時を調べますか？（数値を入れてください）：", "50");
    const targetCount = parseInt(targetCountStr, 10);
    
    if (isNaN(targetCount) || targetCount <= 0) {
        console.log("正しい数値を入力してください。");
        return;
    }

    let allComments = [];
    const commentIds = new Set(); // 重複チェック用のセット
    let offset = 0;
    let hasMore = true;
    const limit = 40; 

    console.log(`スタジオ ${studioId} の全コメントのタイムスタンプを解析中しています...`);

    while (hasMore) {
        try {
            // 親コメントを取得
            const url = `https://${targetDomain}/studios/${studioId}/comments?limit=${limit}&offset=${offset}`;
            const response = await fetch(url);
            if (!response.ok) { hasMore = false; break; }
            const comments = await response.json();

            if (!comments || !Array.isArray(comments) || comments.length === 0) {
                hasMore = false;
                break;
            }

            // 親コメントを配列に保存（重複排除付き）
            for (const comment of comments) {
                if (!commentIds.has(comment.id)) {
                    commentIds.add(comment.id);
                    allComments.push({
                        id: comment.id,
                        user: comment.author.username,
                        text: comment.content,
                        date: new Date(comment.datetime_created)
                    });
                }
            }

            // 返信（replies）を掘り下げる
            for (const comment of comments) {
                let replyOffset = 0;
                let hasMoreReplies = true;

                while (hasMoreReplies) {
                    const replyUrl = `https://${targetDomain}/studios/${studioId}/comments/${comment.id}/replies?limit=${limit}&offset=${replyOffset}`;
                    const replyResponse = await fetch(replyUrl);
                    
                    // エラー時は完全にループを終了させる
                    if (!replyResponse.ok) {
                        hasMoreReplies = false;
                        break;
                    }
                    
                    const replies = await replyResponse.json();

                    if (!replies || !Array.isArray(replies) || replies.length === 0) {
                        hasMoreReplies = false;
                        break;
                    }

                    // 返信も配列に保存（重複排除付き）
                    for (const reply of replies) {
                        if (!commentIds.has(reply.id)) {
                            commentIds.add(reply.id);
                            allComments.push({
                                id: reply.id,
                                user: reply.author.username,
                                text: reply.content,
                                date: new Date(reply.datetime_created)
                            });
                        }
                    }
                    replyOffset += limit;
                    
                    // API規制（429エラー）対策として少し長めに待機
                    await new Promise(resolve => setTimeout(resolve, 200));
                }
            }

            offset += limit;
            // API規制対策として少し長めに待機
            await new Promise(resolve => setTimeout(resolve, 300));

        } catch (error) {
            console.error("エラーが発生しました:", error);
            hasMore = false;
        }
    }

    console.log(`データの収集が完了しました。総件数: ${allComments.length} 件`);

    // 3. すべてのコメントを「投稿日時が古い順（過去 → 現在）」に並び替える
    allComments.sort((b, c) => b.date - c.date);

    // 4. a個目のコメントが存在するかチェックして結果を表示
    if (allComments.length < targetCount) {
        console.log(`%c【判定不可】現在の総コメント数は ${allComments.length} 件なので、まだ ${targetCount} 件には達していません。`, "color: #ff0000; font-weight: bold;");
    } else {
        const targetComment = allComments[targetCount - 1];
        
        // 見やすい日付フォーマットに変換
        const formattedDate = targetComment.date.toLocaleString("ja-JP", {
            year: "numeric",
            month: "2-digit",
            day: "2-digit",
            hour: "2-digit",
            minute: "2-digit",
            second: "2-digit",
            timeZoneName: "short"
        });

        console.log("--------------------------------------------------");
        console.log(`%c【解析結果】`, "color: #00ff00; font-weight: bold; font-size: 16px;");
        console.log(`%cコメント数が「${targetCount}個」に達した瞬間：`, "font-weight: bold; font-size: 14px;");
        console.log(`%c👉 ${formattedDate}`, "color: #ffff00; font-weight: bold; font-size: 16px; background-color: #333; padding: 2px 5px;");
        console.log(`記念すべき ${targetCount} 個目のコメントはこちらです：`);
        console.log(`ユーザー: ${targetComment.user}`);
        console.log(`本文: 「${targetComment.text}」`);
        console.log("--------------------------------------------------");
    }
})();
