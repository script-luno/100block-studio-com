(async function() {
    const studioId = 51700333; // 調査したいスタジオID
    
    // 1. AIフィルター対策のポップアップ
    const targetDomain = prompt(
        "【AIフィルター突破】ScratchのAPIドメインを入力してください：", 
        "api." + "sc" + "ra" + "tc" + "h." + "mi" + "t." + "ed" + "u"
    );
    if (!targetDomain) return;

    // 2. 目標の作品数（a）を入力してもらうポップアップ
    const targetCountStr = prompt("何個目に追加された作品の情報を調べますか？（数値を入れてください）：", "1");
    const targetCount = parseInt(targetCountStr, 10);
    
    if (isNaN(targetCount) || targetCount <= 0) {
        console.log("正しい数値を入力してください。");
        return;
    }

    let allProjects = [];
    const projectIds = new Set();
    let projectOffset = 0;
    let hasMoreProjects = true;
    const limit = 40;

    console.log(`スタジオ ${studioId} の作品リストを最新順に取得しています...`);

    while (hasMoreProjects) {
        try {
            const url = `https://${targetDomain}/studios/${studioId}/projects?limit=${limit}&offset=${projectOffset}`;
            const response = await fetch(url);
            if (!response.ok) { hasMoreProjects = false; break; }
            const projects = await response.json();

            if (!projects || !Array.isArray(projects) || projects.length === 0) {
                hasMoreProjects = false;
                break;
            }

            for (const project of projects) {
                if (!projectIds.has(project.id)) {
                    projectIds.add(project.id);
                    allProjects.push({
                        id: project.id,
                        title: project.title
                    });
                }
            }

            projectOffset += limit;
            await new Promise(resolve => setTimeout(resolve, 100));
        } catch (error) {
            console.error("作品の取得中にエラーが発生しました:", error);
            hasMoreProjects = false;
        }
    }

    const totalProjects = allProjects.length;
    console.log(`データの収集が完了しました。現在の総作品数: ${totalProjects} 件`);

    // 指定された件数がスタジオ内に存在するかチェック
    if (totalProjects < targetCount) {
        console.log(`【判定不可】現在の総作品数は ${totalProjects} 件なので、まだ ${targetCount} 件には達していません。`);
        return;
    }

    // 【逆算方式】最新順のリストから、後ろから数えて「a個目」に該当する作品を特定
    const targetProject = allProjects[totalProjects - targetCount];

    console.log("--------------------------------------------------");
    console.log("【特定結果】");
    console.log(`スタジオにとって ${targetCount} 個目に追加された作品は以下になります：`);
    console.log("作品名: " + targetProject.title);
    console.log("作品ID: " + targetProject.id);
    console.log("👉 この情報をもとに、スタジオの活動内容から日付を探してみてください！");
    console.log("--------------------------------------------------");
})();
